**Goal:** Ship `Esatto.Umbraco.Backoffice.AICosts` 1.0.0 — Esatto.Web's AI → Analytics → Costs tab as a public package, plus cost by profile/user/provider, a "This month" budget tile, CSV export and previous-period comparison.

## Review Focus

1. **A day boundary inside the hourly window for an older range** — the previous period must read Daily for days older than yesterday; if `UsageWindow` were anchored to the range end it would query Hourly 30+ days back (Hourly is kept ~30 days) and show zeros. Pinned in Task 3 (`Previous_period_older_than_yesterday_is_read_from_daily`).
2. **Profile dimension IDs that aren't GUIDs** (deleted/legacy rows) — the breakdown must skip the filter-by-profile for an unparseable ID rather than throw for the whole card. Pinned in Task 5 (`Profile_row_with_a_non_guid_id_is_skipped`).
3. **Budget currency not in the rates** (ECB down, budget in SEK, base USD) — the tile must still show cost and projection, with the reason the budget is missing. Pinned in Task 8 (`budgetInDisplayCurrency` "no rate" case).
4. **CSV values with the separator, quotes or newlines** (a model or user name like `Doe, Jane "JD"`) — must be quoted per RFC 4180 and stay one field. Pinned in Task 8 (`csv` quoting test).
5. **Month projection on the 1st** — one day elapsed must not divide by zero or show an absurd value from a partial day; projection = cost × days in month. Pinned in Task 4 (`Month_projection_on_the_first_day`).

```csharp
namespace Esatto.Umbraco.Backoffice.AICosts;

/// <summary>
/// Bound from <c>Esatto:AICosts</c>. Umbraco.AI records tokens, not money — cost is tokens × these list
/// prices, so it is an estimate that excludes discounts, batch pricing, taxes and provider surcharges.
/// </summary>
public sealed class AICostOptions
{
    public const string SectionName = "Esatto:AICosts";

    /// <summary>Default for <see cref="Currencies"/> when the config section doesn't set any.</summary>
    public static readonly IReadOnlyList<string> DefaultCurrencies = ["SEK", "EUR", "USD", "NOK", "DKK", "GBP"];

    /// <summary>Currency the prices below are quoted in (ISO 4217). Provider list prices are in USD.</summary>
    public string BaseCurrency { get; set; } = "USD";

    /// <summary>Currencies offered in the currency switcher (ISO 4217, ECB-quoted). Deliberately not
    /// pre-filled: the configuration binder appends to an existing list, so an initialiser here would
    /// duplicate every configured currency. See <see cref="DefaultCurrencies"/>.</summary>
    public List<string> Currencies { get; set; } = [];

    /// <summary>Currency the tab opens in.</summary>
    public string DefaultCurrency { get; set; } = "SEK";

    /// <summary>Where the prices came from and when they were checked — shown in the footnote.</summary>
    public string? PricesSource { get; set; }

    /// <summary>Optional budget per calendar month (UTC), for the "This month" tile.</summary>
    public decimal? MonthlyBudget { get; set; }

    /// <summary>Currency of <see cref="MonthlyBudget"/>; the default currency when not set.</summary>
    public string? BudgetCurrency { get; set; }

    /// <summary>Prices per 1 million tokens in <see cref="BaseCurrency"/>, keyed by model ID exactly as
    /// Umbraco.AI reports it (e.g. <c>claude-sonnet-5-5</c>). Models without an entry show tokens but no cost.</summary>
    public Dictionary<string, AIModelPrice> PricesPerMillionTokens { get; set; } = new(StringComparer.OrdinalIgnoreCase);
}

/// <summary>List price per 1 million tokens. <see cref="CachedInput"/> is the cache-read price;
/// when omitted, cached tokens are charged at the <see cref="Input"/> price (never under-estimates).</summary>
public sealed class AIModelPrice
{
    public decimal Input { get; set; }
    public decimal Output { get; set; }
    public decimal? CachedInput { get; set; }
}
```

```csharp
namespace Esatto.Umbraco.Backoffice.AICosts;

public static class AICostCalculator
{
    private const decimal TokensPerUnit = 1_000_000m;

    /// <summary>
    /// Estimated cost in the price's currency. Cached input tokens are a subset of input tokens (the
    /// Microsoft.Extensions.AI convention): the cached part is billed at <see cref="AIModelPrice.CachedInput"/>
    /// and only the remainder at the full input price. Negative counts count as zero.
    /// </summary>
    public static decimal Cost(AIModelPrice price, long inputTokens, long outputTokens, long cachedInputTokens)
    {
        var input = Math.Max(inputTokens, 0);
        var cached = Math.Clamp(cachedInputTokens, 0, input);

        return ((input - cached) * price.Input
                + cached * (price.CachedInput ?? price.Input)
                + Math.Max(outputTokens, 0) * price.Output) / TokensPerUnit;
    }
}
```

```csharp
namespace Esatto.Umbraco.Backoffice.AICosts;

/// <summary>
/// Which Umbraco.AI statistics cover a time range. Umbraco.AI rolls hourly statistics into daily ones
/// only for completed days (up to an hour after midnight UTC) and keeps hourly statistics ~30 days, so:
/// <b>Daily</b> before yesterday, <b>Hourly</b> for yesterday and today. Anchored to <i>now</i>, not to
/// the range end, so an older range (the previous period) is read from Daily only.
/// </summary>
public sealed record UsageWindow(DateTime From, DateTime To, DateTime HourlyFrom)
{
    public static UsageWindow Create(DateTime from, DateTime to, DateTime utcNow)
    {
        var yesterday = DateTime.SpecifyKind(utcNow.Date.AddDays(-1), DateTimeKind.Utc);
        var hourlyFrom = yesterday < from ? from : yesterday > to ? to : yesterday;
        return new UsageWindow(from, to, hourlyFrom);
    }

    public bool HasDaily => From < HourlyFrom;
    public bool HasHourly => HourlyFrom < To;
}
```

```csharp
public sealed record UsageTotals(long Requests, long InputTokens, long OutputTokens, long CachedInputTokens, long TotalTokens)
{
    public static readonly UsageTotals Empty = new(0, 0, 0, 0, 0);

    public UsageTotals Plus(UsageTotals o) => new(
        Requests + o.Requests, InputTokens + o.InputTokens, OutputTokens + o.OutputTokens,
        CachedInputTokens + o.CachedInputTokens, TotalTokens + o.TotalTokens);
}

/// <param name="PricedTokens">Tokens from models with a configured price.</param>
public sealed record PricedUsage(UsageTotals Totals, decimal Cost, long PricedTokens, long UnpricedTokens);

/// <summary>Reads Umbraco.AI's usage statistics for a <see cref="UsageWindow"/> (Daily + Hourly parts).</summary>
public sealed class AICostQueries
{
    private readonly IAIUsageAnalyticsService _analytics;

    public AICostQueries(IAIUsageAnalyticsService analytics) => _analytics = analytics;

    public async Task<List<AIUsageTimeSeriesPoint>> GetSeriesAsync(UsageWindow window, AIUsageFilter filter, CancellationToken ct)
    {
        var points = new List<AIUsageTimeSeriesPoint>();
        if (window.HasDaily)
            points.AddRange(await _analytics.GetTimeSeriesAsync(window.From, window.HourlyFrom, AIUsagePeriod.Daily, filter, ct));
        if (window.HasHourly)
            points.AddRange(await _analytics.GetTimeSeriesAsync(window.HourlyFrom, window.To, AIUsagePeriod.Hourly, filter, ct));
        return points;
    }

    public async Task<IReadOnlyList<string>> GetModelIdsAsync(UsageWindow window, CancellationToken ct) =>
        (await GetItemsAsync(window, (from, to, period) => _analytics.GetBreakdownByModelAsync(from, to, period, ct)))
            .Select(i => i.Dimension).ToList();

    public Task<IReadOnlyList<AIUsageBreakdownItem>> GetBreakdownAsync(UsageWindow window, AICostDimension dimension, CancellationToken ct) =>
        GetItemsAsync(window, dimension switch
        {
            AICostDimension.Profile => (from, to, period) => _analytics.GetBreakdownByProfileAsync(from, to, period, ct),
            AICostDimension.User => (from, to, period) => _analytics.GetBreakdownByUserAsync(from, to, period, ct),
            AICostDimension.Provider => (from, to, period) => _analytics.GetBreakdownByProviderAsync(from, to, period, ct),
            _ => throw new ArgumentOutOfRangeException(nameof(dimension)),
        });

    public async Task<UsageTotals> GetTotalsAsync(UsageWindow window, AIUsageFilter filter, CancellationToken ct)
    {
        var totals = UsageTotals.Empty;
        if (window.HasDaily)
            totals = totals.Plus(ToTotals(await _analytics.GetSummaryAsync(window.From, window.HourlyFrom, AIUsagePeriod.Daily, filter, ct)));
        if (window.HasHourly)
            totals = totals.Plus(ToTotals(await _analytics.GetSummaryAsync(window.HourlyFrom, window.To, AIUsagePeriod.Hourly, filter, ct)));
        return totals;
    }

    /// <summary>
    /// Totals per model, each priced with <see cref="AICostCalculator"/>. Pricing is linear in tokens, so
    /// pricing a model's totals equals pricing each of its data points.
    /// </summary>
    public async Task<PricedUsage> PriceByModelAsync(
        UsageWindow window, IReadOnlyList<string> modelIds, Func<string, AIUsageFilter> filterForModel,
        IReadOnlyDictionary<string, AIModelPrice> prices, CancellationToken ct)
    {
        var totals = UsageTotals.Empty;
        decimal cost = 0;
        long priced = 0, unpriced = 0;
        foreach (var modelId in modelIds)
        {
            var model = await GetTotalsAsync(window, filterForModel(modelId), ct);
            totals = totals.Plus(model);
            if (prices.TryGetValue(modelId, out var price))
            {
                cost += AICostCalculator.Cost(price, model.InputTokens, model.OutputTokens, model.CachedInputTokens);
                priced += model.TotalTokens;
            }
            else
            {
                unpriced += model.TotalTokens;
            }
        }
        return new PricedUsage(totals, cost, priced, unpriced);
    }

    // Daily + Hourly breakdown rows merged by dimension; the first non-empty display name wins.
    private static async Task<IReadOnlyList<AIUsageBreakdownItem>> GetItemsAsync(
        UsageWindow window, Func<DateTime, DateTime, AIUsagePeriod, Task<IEnumerable<AIUsageBreakdownItem>>> query)
    {
        var items = new List<AIUsageBreakdownItem>();
        if (window.HasDaily) items.AddRange(await query(window.From, window.HourlyFrom, AIUsagePeriod.Daily));
        if (window.HasHourly) items.AddRange(await query(window.HourlyFrom, window.To, AIUsagePeriod.Hourly));

        return items
            .GroupBy(i => i.Dimension, StringComparer.Ordinal)
            .Select(g => g.FirstOrDefault(i => !string.IsNullOrEmpty(i.DimensionName)) ?? g.First())
            .ToList();
    }

    private static UsageTotals ToTotals(AIUsageSummary s) =>
        new(s.TotalRequests, s.InputTokens, s.OutputTokens, s.CachedInputTokens ?? 0, s.TotalTokens);
}
```

```csharp
    private async Task<AICostMonth> GetMonthAsync(DateOnly today, DateTime now, AICostOptions options, string defaultCurrency, CancellationToken ct)

    // The same number of days right before the selected period — only when Umbraco.AI still has all of
    // it: daily statistics older than UsageDailyRetentionDays are deleted, and a comparison against
    // partly deleted data would mislead.
    private async Task<AICostPrevious?> GetPreviousAsync(DateOnly firstDay, int days, DateOnly today, DateTime now, AICostOptions options, CancellationToken ct)
    {
        var previousFrom = firstDay.AddDays(-days);
        if (previousFrom < today.AddDays(-_analyticsOptions.Value.UsageDailyRetentionDays)) return null;

        var window = UsageWindow.Create(previousFrom.ToDateTime(TimeOnly.MinValue, DateTimeKind.Utc),
            firstDay.ToDateTime(TimeOnly.MinValue, DateTimeKind.Utc), now);
        var priced = await PriceAsync(window, options, ct);
        return new AICostPrevious(previousFrom, firstDay.AddDays(-1),
            new AICostTotals(priced.Totals.Requests, priced.Totals.TotalTokens, priced.Cost, priced.UnpricedTokens));
    }

    private async Task<PricedUsage> PriceAsync(UsageWindow window, AICostOptions options, CancellationToken ct) =>
        await _queries.PriceByModelAsync(window, await _queries.GetModelIdsAsync(window, ct),
            modelId => new AIUsageFilter { ModelId = modelId }, options.PricesPerMillionTokens, ct);
```

```csharp
using Microsoft.Extensions.Options;
using Umbraco.AI.Core.Analytics.Usage;

namespace Esatto.Umbraco.Backoffice.AICosts;

/// <summary>
/// Cost by profile, user or provider. Cost depends on the model, so each row is priced from its usage
/// per model used in the period (rows × models summary queries; it only runs when an editor opens it).
/// </summary>
public sealed class AICostBreakdownService
{
    private readonly AICostQueries _queries;
    private readonly IOptionsMonitor<AICostOptions> _options;
    private readonly TimeProvider _timeProvider;

    public AICostBreakdownService(AICostQueries queries, IOptionsMonitor<AICostOptions> options, TimeProvider timeProvider)
    {
        _queries = queries;
        _options = options;
        _timeProvider = timeProvider;
    }

    public async Task<AICostBreakdown> GetBreakdownAsync(AICostDimension dimension, int days, CancellationToken ct)
    {
        days = Math.Clamp(days, 1, AICostReportService.MaxDays);
        var now = _timeProvider.GetUtcNow().UtcDateTime;
        var today = DateOnly.FromDateTime(now);
        var window = UsageWindow.Create(
            today.AddDays(-(days - 1)).ToDateTime(TimeOnly.MinValue, DateTimeKind.Utc),
            today.AddDays(1).ToDateTime(TimeOnly.MinValue, DateTimeKind.Utc), now);
        var prices = _options.CurrentValue.PricesPerMillionTokens;
        var modelIds = await _queries.GetModelIdsAsync(window, ct);

        var rows = new List<AICostBreakdownRow>();
        foreach (var item in await _queries.GetBreakdownAsync(window, dimension, ct))
        {
            if (Filter(dimension, item.Dimension) is not { } rowFilter) continue;

            var priced = await _queries.PriceByModelAsync(window, modelIds, modelId => rowFilter with { ModelId = modelId }, prices, ct);
            var t = priced.Totals;
            rows.Add(new AICostBreakdownRow(
                item.Dimension,
                string.IsNullOrWhiteSpace(item.DimensionName) ? item.Dimension : item.DimensionName,
                t.Requests, t.InputTokens, t.CachedInputTokens, t.OutputTokens, t.TotalTokens,
                priced.PricedTokens == 0 && priced.UnpricedTokens > 0 ? null : priced.Cost,
                priced.UnpricedTokens));
        }

        return new AICostBreakdown(
            dimension.ToString().ToLowerInvariant(),
            rows.OrderByDescending(r => r.Cost ?? -1).ThenByDescending(r => r.TotalTokens).ToList());
    }

    // A profile row's ID is the profile's GUID; one that isn't (a deleted or legacy row) can't be
    // filtered on, so it's left out rather than failing the whole card.
    private static AIUsageFilter? Filter(AICostDimension dimension, string id) => dimension switch
    {
        AICostDimension.Profile => Guid.TryParse(id, out var profileId) ? new AIUsageFilter { ProfileId = profileId } : null,
        AICostDimension.User => new AIUsageFilter { UserId = id },
        AICostDimension.Provider => new AIUsageFilter { ProviderId = id },
        _ => null,
    };
}
```

```csharp
        if (services.Any(d => d.ServiceType == typeof(AICostQueries))) return services;

        services.AddScoped<AICostQueries>();

        services.AddScoped<AICostBreakdownService>();
        return services;
    }
```

```ts
import type { AICostReport } from './api';
import { convert } from './money';

export type BudgetState =
  | { kind: 'none' }
  | { kind: 'unavailable'; reason: string }
  | { kind: 'set'; amount: number; spentShare: number; level: 'ok' | 'warning' | 'over' };

/** The month's budget in the display currency, and how far the month is into it. */
export function budgetInDisplayCurrency(report: AICostReport, currency: string): BudgetState {
  const budget = report.month.budget;
  if (!budget) return { kind: 'none' };

  // Budget → base currency (the month's cost is in base), via the budget currency's rate.
  const rates = report.rates?.perBaseUnit;
  const budgetRate = budget.currency === report.baseCurrency ? 1 : rates?.[budget.currency];
  if (!budgetRate) {
    return { kind: 'unavailable', reason: `Exchange rates are unavailable, so the ${budget.currency} budget can’t be compared right now.` };
  }
  const budgetInBase = budget.amount / budgetRate;
  const spentShare = budgetInBase > 0 ? report.month.cost / budgetInBase : 0;
  const level = report.month.cost > budgetInBase ? 'over' : spentShare >= 0.8 || report.month.projectedCost > budgetInBase ? 'warning' : 'ok';
  return { kind: 'set', amount: convert(report, budgetInBase, currency), spentShare, level };
}
```

```ts
export type Change = { kind: 'none' } | { kind: 'new' } | { kind: 'change'; percent: number };

/** Change vs the previous period; "new" when there was nothing before. Rounded to whole percent. */
export function change(current: number, previous: number | undefined | null): Change {
  if (previous === null || previous === undefined) return { kind: 'none' };
  if (previous === 0) return current === 0 ? { kind: 'change', percent: 0 } : { kind: 'new' };
  return { kind: 'change', percent: Math.round(((current - previous) / previous) * 100) };
}
```

```ts
/** Excel splits CSV on the locale's list separator: ';' where the decimal separator is ','. */
export function csvSeparators(locale?: string): { field: ';' | ','; decimal: ',' | '.' } {
  const decimal = new Intl.NumberFormat(locale).formatToParts(1.5).find((p) => p.type === 'decimal')?.value;
  return decimal === ',' ? { field: ';', decimal: ',' } : { field: ',', decimal: '.' };
}

/** RFC 4180 CSV with a UTF-8 BOM (so Excel reads å/ä/ö), numbers in the locale's decimal style, no grouping. */
export function toCsv(header: string[], rows: (string | number)[][], locale?: string): string {
  const { field, decimal } = csvSeparators(locale);
  const cell = (v: string | number) => {
    const text = typeof v === 'number' ? String(v).replace('.', decimal) : v;
    return /["\r\n]/.test(text) || text.includes(field) ? `"${text.replace(/"/g, '""')}"` : text;
  };
  return '\uFEFF' + [header, ...rows].map((r) => r.map(cell).join(field)).join('\r\n');
}

export function downloadCsv(fileName: string, text: string): void {
  const url = URL.createObjectURL(new Blob([text], { type: 'text/csv;charset=utf-8' }));
  const a = Object.assign(document.createElement('a'), { href: url, download: fileName });
  a.click();
  URL.revokeObjectURL(url);
}
```
